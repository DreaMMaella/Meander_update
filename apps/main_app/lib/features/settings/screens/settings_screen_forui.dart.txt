/// Экспериментальная версия экрана настроек на [forui] (оценка библиотеки).
/// Бизнес-логика — 1-в-1 с `settings_screen.dart`, UI полностью на forui:
/// `FTheme`/`FScaffold`/`FHeader`/`FTileGroup`/`FTile`/`FSwitch`/
/// `FSelectMenuTile`/`FDialog`/`FToast`/`FButton`.
class SettingsScreenForui extends ConsumerStatefulWidget {
  const SettingsScreenForui({super.key});

  @override
  ConsumerState<SettingsScreenForui> createState() =>
      _SettingsScreenForuiState();
}

class _SettingsScreenForuiState extends ConsumerState<SettingsScreenForui> {
  String? _currentUserId;

  /// Контекст ниже `FToaster` — для `showFToast` из обработчиков/диалогов.
  late BuildContext _contentContext;

  @override
  void initState() {
    super.initState();
    _loadUser();
  }

  Future<void> _loadUser() async {
    final prefs = await SharedPreferences.getInstance();
    if (mounted) {
      setState(() {
        _currentUserId = prefs.getString('user_id');
      });
    }
    _syncTimezone();
  }

  Future<void> _syncTimezone() async {
    if (_currentUserId == null) return;
    try {
      final prefs = await SharedPreferences.getInstance();
      final lastSync = prefs.getString('tz_last_sync');
      final today = DateTime.now().toIso8601String().substring(0, 10);
      if (lastSync == today) return;

      final offset = DateTime.now().timeZoneOffset.inMinutes;
      await ref.read(apiServiceProvider).post('profiles/me', body: {
        'timezone_offset': offset,
      });
      await prefs.setString('tz_last_sync', today);
      ref.invalidate(userProfileProvider(_currentUserId!));
    } catch (_) {}
  }

  void _toast(
    String title, {
    String? description,
    FToastVariant variant = FToastVariant.primary,
    Widget? icon,
  }) => showFToast(
    context: _contentContext,
    title: Text(title),
    description: description == null ? null : Text(description),
    icon: icon,
    variant: variant,
  );

  Future<void> _toggleQuietHours(bool current) async {
    if (_currentUserId == null) return;
    try {
      await ref.read(apiServiceProvider).post(
        'profiles/me',
        body: {'quiet_hours_enabled': !current},
      );
      ref.invalidate(userProfileProvider(_currentUserId!));
    } catch (_) {}
  }

  Future<void> _handleLogout() async {
    final confirm = await _confirmDialog(
      title: 'Выйти из аккаунта?',
      body: 'Вы уверены, что хотите выйти?',
      confirmLabel: 'Выйти',
    );

    if (confirm == true && _currentUserId != null) {
      final prefs = await SharedPreferences.getInstance();
      await prefs.remove('jwt_token');
      await prefs.remove('user_id');

      ref.invalidate(communityFeedProvider);
      ref.invalidate(communityOnlineProvider);
      ref.invalidate(communityTopProvider);

      if (mounted) {
        Navigator.pushAndRemoveUntil(
          context,
          MaterialPageRoute(builder: (_) => const MenuScreen()),
          (route) => false,
        );
      }
    }
  }

  Future<void> _copyProfileId() async {
    if (_currentUserId == null) return;
    await Clipboard.setData(ClipboardData(text: _currentUserId!));
    _toast('ID скопирован', icon: const Icon(FLucideIcons.copy));
  }

  void _showDebugUpdateDialog(BuildContext context) {
    final changelog = [
      '• Новый дизайн окна обновления',
      '• Исправлен pinch-to-zoom на канвасе',
      '• Исправлено выделение текста в редакторе',
      '• Плеер квестов вынесен в отдельный модуль mnd_player',
      '• Добавлен template_quest_app для сборки standalone квестов',
      '• Сохранения через mnd_core SaveStore',
    ].join('\n');

    Navigator.of(context).push(
      MaterialPageRoute(
        builder: (_) => UpdateScreen(
          version: '99.0.0',
          changelog: changelog,
          downloadUrl: '',
          isCritical: false,
        ),
      ),
    );
  }

  Future<void> _editName() async {
    if (_currentUserId == null) return;
    final profileAsync = ref.read(userProfileProvider(_currentUserId!));
    final currentName = profileAsync.maybeWhen(
      data: (p) => p.fullName,
      orElse: () => '',
    );
    final newName = await _promptDialog(
      title: 'Изменить имя',
      hint: 'Новое имя',
      initial: currentName,
    );
    if (newName == null || newName.isEmpty || newName == currentName) return;
    try {
      await ref
          .read(apiServiceProvider)
          .post('profiles/me', body: {'full_name': newName});
      ref.invalidate(userProfileProvider(_currentUserId!));
      _toast('Имя обновлено');
    } catch (e) {
      _toast('Ошибка', description: '$e', variant: FToastVariant.destructive);
    }
  }

  Future<void> _toggleNsfw(bool current) async {
    if (_currentUserId == null) return;

    if (!current) {
      final confirm = await _confirmDialog(
        title: 'Внимание',
        body:
            'Контент 18+ может содержать нецензурную лексику, жестокость '
            'и материалы для взрослых.\n\nПодтвердите, что вам исполнилось '
            '18 лет.',
        confirmLabel: 'Мне есть 18 лет',
      );
      if (confirm != true) return;
    }

    try {
      await ref
          .read(apiServiceProvider)
          .post('profiles/me', body: {'show_nsfw': !current});
      ref.invalidate(userProfileProvider(_currentUserId!));
      _toast(current ? 'Контент 18+ скрыт' : 'Контент 18+ включен');
    } catch (e) {
      _toast('Ошибка', description: '$e', variant: FToastVariant.destructive);
    }
  }

  Future<void> _editBio() async {
    if (_currentUserId == null) return;
    final profileAsync = ref.read(userProfileProvider(_currentUserId!));
    final currentBio = profileAsync.maybeWhen(
      data: (p) => p.bio ?? '',
      orElse: () => '',
    );
    final newBio = await _promptDialog(
      title: 'О себе',
      hint: 'Расскажите о себе…',
      initial: currentBio,
      maxLines: 4,
      maxLength: 280,
    );
    if (newBio == null || newBio == currentBio) return;
    try {
      await ref
          .read(apiServiceProvider)
          .post('profiles/me', body: {'bio': newBio});
      ref.invalidate(userProfileProvider(_currentUserId!));
      _toast('Описание обновлено');
    } catch (e) {
      _toast('Ошибка', description: '$e', variant: FToastVariant.destructive);
    }
  }

  Future<void> _showRedeemCodeDialog() async {
    final result = await _promptDialog(
      title: 'Ввести код',
      hint: 'Введите промокод',
      confirmLabel: 'Активировать',
    );
    if (result != null && result.isNotEmpty) {
      // TODO: implement redeem API call
      _toast('Код активирован');
    }
  }

  // ────────────────────────────────────────────────────────────────────────
  //  Диалоги (forui)
  // ────────────────────────────────────────────────────────────────────────

  Future<bool?> _confirmDialog({
    required String title,
    required String body,
    required String confirmLabel,
  }) => showFDialog<bool>(
    context: _contentContext,
    builder: (context, style, _) => Padding(
      padding: const EdgeInsets.all(20),
      child: Column(
        mainAxisSize: MainAxisSize.min,
        crossAxisAlignment: CrossAxisAlignment.start,
        children: [
          Text(title, style: style.titleTextStyle),
          const SizedBox(height: 10),
          Text(body, style: style.bodyTextStyle),
          const SizedBox(height: 24),
          Row(
            children: [
              Expanded(
                child: FButton(
                  variant: FButtonVariant.outline,
                  size: FButtonSizeVariant.lg,
                  onPress: () => Navigator.pop(context, false),
                  child: const Text('Отмена'),
                ),
              ),
              const SizedBox(width: 10),
              Expanded(
                child: FButton(
                  variant: FButtonVariant.destructive,
                  size: FButtonSizeVariant.lg,
                  onPress: () => Navigator.pop(context, true),
                  child: Text(confirmLabel),
                ),
              ),
            ],
          ),
        ],
      ),
    ),
  );

  Future<String?> _promptDialog({
    required String title,
    required String hint,
    String initial = '',
    int maxLines = 1,
    int? maxLength,
    String confirmLabel = 'Сохранить',
  }) {
    final controller = TextEditingController(text: initial);
    return showFDialog<String>(
      context: _contentContext,
      builder: (context, style, _) => Padding(
        padding: const EdgeInsets.all(20),
        child: Column(
          mainAxisSize: MainAxisSize.min,
          crossAxisAlignment: CrossAxisAlignment.start,
          children: [
            Text(title, style: style.titleTextStyle),
            const SizedBox(height: 16),
            FTextField(
              control: FTextFieldControl.managed(controller: controller),
              hint: hint,
              autofocus: true,
              maxLines: maxLines,
              maxLength: maxLength,
            ),
            const SizedBox(height: 24),
            Row(
              children: [
                Expanded(
                  child: FButton(
                    variant: FButtonVariant.outline,
                    size: FButtonSizeVariant.lg,
                    onPress: () => Navigator.pop(context),
                    child: const Text('Отмена'),
                  ),
                ),
                const SizedBox(width: 10),
                Expanded(
                  child: FButton(
                    size: FButtonSizeVariant.lg,
                    onPress: () => Navigator.pop(context, controller.text.trim()),
                    child: Text(confirmLabel),
                  ),
                ),
              ],
            ),
          ],
        ),
      ),
    );
  }

  // ────────────────────────────────────────────────────────────────────────
  //  Тема forui: синхронизация с ThemeMode приложения
  // ────────────────────────────────────────────────────────────────────────

  FThemeData _resolveTheme(BuildContext context, ThemeMode mode) {
    final brightness = switch (mode) {
      ThemeMode.light => Brightness.light,
      ThemeMode.dark => Brightness.dark,
      ThemeMode.system => MediaQuery.platformBrightnessOf(context),
    };
    return brightness == Brightness.dark
        ? FTheme.neutral.dark.touch
        : FTheme.neutral.light.touch;
  }

  @override
  Widget build(BuildContext context) {
    final themeMode = ref.watch(themeModeProvider);
    final theme = _resolveTheme(context, themeMode);

    final isLoggedIn = _currentUserId != null;
    final isAdmin = isLoggedIn
        ? ref
              .watch(userProfileProvider(_currentUserId!))
              .maybeWhen(data: (p) => p.role == 'admin', orElse: () => false)
        : false;

    return FTheme(
      data: theme,
      child: Material(
        color: theme.colors.background,
        child: FToaster(
          child: FScaffold(
            childPad: false,
            header: FHeader(title: const Text('Настройки')),
            child: Center(
              child: ConstrainedBox(
                constraints: const BoxConstraints(maxWidth: 720),
                child: Builder(
                  builder: (contentContext) {
                    _contentContext = contentContext;
                    return SingleChildScrollView(
                      physics: const BouncingScrollPhysics(
                        parent: AlwaysScrollableScrollPhysics(),
                      ),
                      padding: const EdgeInsets.symmetric(horizontal: 16)
                          .copyWith(bottom: 96),
                      child: Column(
                        crossAxisAlignment: CrossAxisAlignment.stretch,
                        children: [
                          if (isLoggedIn)
                            Consumer(
                              builder: (context, ref, _) {
                                final profileAsync = ref.watch(
                                  userProfileProvider(_currentUserId!),
                                );
                                final name = profileAsync.maybeWhen(
                                  data: (p) => p.fullName,
                                  orElse: () => null,
                                );
                                final bio = profileAsync.maybeWhen(
                                  data: (p) =>
                                      (p.bio == null || p.bio!.isEmpty)
                                      ? null
                                      : p.bio,
                                  orElse: () => null,
                                );
                                final bool showNsfw = profileAsync.maybeWhen(
                                  data: (p) => p.showNsfw,
                                  orElse: () => false,
                                );

                                return FTileGroup(
                                  label: const Text('Профиль'),
                                  children: [
                                    FTile(
                                      title: const Text('Редактировать имя'),
                                      subtitle: name == null
                                          ? null
                                          : Text(name),
                                      prefix: FAvatar.raw(
                                        size: 36,
                                        child: const Icon(FLucideIcons.userRound),
                                      ),
                                      suffix: const Icon(FLucideIcons.chevronRight),
                                      onPress: _editName,
                                    ),
                                    FTile(
                                      title: const Text('Редактировать био'),
                                      subtitle: Text(bio ?? 'Не задано'),
                                      prefix: const Icon(FLucideIcons.fileText),
                                      suffix: const Icon(FLucideIcons.chevronRight),
                                      onPress: _editBio,
                                    ),
                                    FTile(
                                      variant: showNsfw
                                          ? FItemVariant.destructive
                                          : FItemVariant.primary,
                                      title: const Text('Контент 18+'),
                                      subtitle: Text(
                                        showNsfw
                                            ? 'Включен'
                                            : 'Скрыт из ленты',
                                      ),
                                      prefix: Icon(
                                        showNsfw
                                            ? FLucideIcons.eye
                                            : FLucideIcons.eyeOff,
                                      ),
                                      suffix: FSwitch(
                                        value: showNsfw,
                                        onChange: (_) =>
                                            _toggleNsfw(showNsfw),
                                      ),
                                    ),
                                  ],
                                );
                              },
                            ),

                          FTileGroup(
                            label: const Text('Внешний вид'),
                            children: [
                              FSelectMenuTile<ThemeMode>.fromMap(
                                const {
                                  'Как в системе': ThemeMode.system,
                                  'Светлая': ThemeMode.light,
                                  'Тёмная': ThemeMode.dark,
                                },
                                title: const Text('Режим темы'),
                                prefix: const Icon(FLucideIcons.sunMoon),
                                selectControl: FMultiValueControl.lifted(
                                  value: {themeMode},
                                  onChange: (selection) {
                                    if (selection.isNotEmpty) {
                                      ref
                                          .read(themeModeProvider.notifier)
                                          .setTheme(selection.first);
                                    }
                                  },
                                ),
                              ),
                              FTile(
                                title: const Text('Тема оформления'),
                                subtitle: const Text('Цвет, фон, пресеты'),
                                prefix: const Icon(FLucideIcons.palette),
                                suffix: const Icon(FLucideIcons.chevronRight),
                                onPress: () => Navigator.push(
                                  context,
                                  MaterialPageRoute(
                                    builder: (_) => const ThemeSettingsScreen(),
                                  ),
                                ),
                              ),
                              FTile(
                                title: const Text('Плагины'),
                                subtitle: const Text(
                                  'Управление расширениями',
                                ),
                                prefix: const Icon(FLucideIcons.puzzle),
                                suffix: const Icon(FLucideIcons.chevronRight),
                                onPress: () => Navigator.push(
                                  context,
                                  MaterialPageRoute(
                                    builder: (_) => const PluginManagerScreen(),
                                  ),
                                ),
                              ),
                            ],
                          ),

                          if (isLoggedIn)
                            Consumer(
                              builder: (context, ref, _) {
                                final notificationsEnabled = ref.watch(
                                  notificationsSettingsProvider,
                                );
                                final profileAsync = ref.watch(
                                  userProfileProvider(_currentUserId!),
                                );
                                final quietHoursEnabled = profileAsync.maybeWhen(
                                  data: (p) => p.quietHoursEnabled,
                                  orElse: () => true,
                                );

                                return FTileGroup(
                                  label: const Text('Уведомления'),
                                  children: [
                                    FTile(
                                      title: const Text('Push-уведомления'),
                                      subtitle: Text(
                                        notificationsEnabled
                                            ? 'Включены'
                                            : 'Выключены',
                                      ),
                                      prefix: Icon(
                                        notificationsEnabled
                                            ? FLucideIcons.bellRing
                                            : FLucideIcons.bellOff,
                                      ),
                                      suffix: FSwitch(
                                        value: notificationsEnabled,
                                        onChange: (value) => ref
                                            .read(
                                              notificationsSettingsProvider
                                                  .notifier,
                                            )
                                            .toggle(value),
                                      ),
                                    ),
                                    FTile(
                                      title: const Text('Не беспокоить ночью'),
                                      subtitle: Text(
                                        quietHoursEnabled
                                            ? 'Push выключены с 22:00 до 08:00'
                                            : 'Пуши приходят в любое время',
                                      ),
                                      prefix: const Icon(FLucideIcons.moonStar),
                                      suffix: FSwitch(
                                        value: quietHoursEnabled,
                                        onChange: (_) =>
                                            _toggleQuietHours(quietHoursEnabled),
                                      ),
                                    ),
                                  ],
                                );
                              },
                            ),

                          FTileGroup(
                            label: const Text('Информация'),
                            children: [
                              if (isLoggedIn)
                                FTile(
                                  title: const Text('Ввести код'),
                                  subtitle: const Text('Активация промокода'),
                                  prefix: const Icon(FLucideIcons.ticket),
                                  suffix: const Icon(FLucideIcons.chevronRight),
                                  onPress: _showRedeemCodeDialog,
                                ),
                              FTile(
                                title: const Text('О нас'),
                                subtitle: const Text(
                                  'История версий, команда',
                                ),
                                prefix: const Icon(FLucideIcons.bookOpen),
                                suffix: const Icon(FLucideIcons.chevronRight),
                                onPress: () => Navigator.push(
                                  context,
                                  MaterialPageRoute(
                                    builder: (_) => const AboutScreen(),
                                  ),
                                ),
                              ),
                              if (isLoggedIn)
                                FTile(
                                  title: const Text('Скопировать ID'),
                                  subtitle: Text(_currentUserId!),
                                  prefix: const Icon(FLucideIcons.copy),
                                  onPress: _copyProfileId,
                                ),
                            ],
                          ),

                          if (kDebugMode)
                            FTileGroup(
                              label: const Text('Разработка'),
                              children: [
                                FTile(
                                  title: const Text('Диалог обновления'),
                                  subtitle: const Text(
                                    'Показать тестовое окно с ченджлогом',
                                  ),
                                  prefix: const Icon(FLucideIcons.bug),
                                  suffix: const Icon(FLucideIcons.chevronRight),
                                  onPress: () =>
                                      _showDebugUpdateDialog(contentContext),
                                ),
                              ],
                            ),

                          if (isAdmin)
                            FTileGroup(
                              label: const Text('Администрирование'),
                              children: [
                                FTile(
                                  variant: FItemVariant.destructive,
                                  title: const Text('Админ-панель'),
                                  subtitle: const Text(
                                    'Модерация, отчёты, блокировки',
                                  ),
                                  prefix: const Icon(FLucideIcons.shieldCheck),
                                  suffix: const Icon(FLucideIcons.chevronRight),
                                  onPress: () => Navigator.push(
                                    contentContext,
                                    MaterialPageRoute(
                                      builder: (_) => const AdminScreen(),
                                    ),
                                  ),
                                ),
                              ],
                            ),

                          if (isLoggedIn) ...[
                            const SizedBox(height: 28),
                            FButton(
                              variant: FButtonVariant.destructive,
                              size: FButtonSizeVariant.lg,
                              prefix: const Icon(FLucideIcons.logOut),
                              onPress: _handleLogout,
                              child: const Text('Выйти из аккаунта'),
                            ),
                          ],
                        ],
                      ),
                    );
                  },
                ),
              ),
            ),
          ),
        ),
      ),
    );
  }
}
